<p>Ho analizzato la struttura completa del tuo file Excel (materie prime con scarti e costi netti, ricette dettagliate delle pizze con relative grammature, matrice del Menù Engineering con margini e popolarità, e prezzi di vendita).</p>
<p>Come richiesto, ho progettato un'<strong>applicazione web completa in un unico file HTML</strong> (stile "Single Page Application"), pronta all'uso e utilizzabile direttamente in qualsiasi browser. Ho integrato <strong>Chart.js</strong> per i grafici dinamici, <strong>SweetAlert2</strong> per i pop-up di sicurezza e conferma intuitivi per utenti non esperti, e un layout responsive curato nei minimi dettagli (Dark / Modern Food Navy Style).</p>

<h3>Caratteristiche dell'applicazione integrata:</h3>
<ul>
  <li><strong>Dashboard Grafica:</strong> Grafici a barre e torta per visualizzare la matrice del Menù Engineering (Star, Dog, Workhorse, Puzzle), il bilanciamento del menù e i top 5 prodotti per impatto sul Food Cost.</li>

            // Prezzo Lordo al kg/l = prezzoLordo / formato

            // Prezzo Netto tenendo conto dello scarto
            return lordoUnita / (1 - (mp.scarto / 100));

            const mp = materiePrime.find(m => m.id === mpId);

            const costoAlKg = getCostoNettoAlKgL(mp);
            return (costoAlKg / 1000) * qtaGrammi;
        }

        function renderRicettaPizza() {
            const pizza = pizze.find(p => p.id === selectedPizzaId);
            const cardSummary = document.getElementById('pizza-summary-card');
            const tbody = document.getElementById('tbl-ricetta-body');
            tbody.innerHTML = '';

            if (!pizza) {
                cardSummary.classList.add('hidden');
                return;
            }

            cardSummary.classList.remove('hidden');
            const costoTotale = getCostoTotalePizza(pizza);
            const margine = pizza.prezzoVendita - costoTotale;
            const fcPerc = pizza.prezzoVendita > 0 ? (costoTotale / pizza.prezzoVendita) * 100 : 0;

            document.getElementById('summary-pizza-nome').innerText = pizza.nome;
            document.getElementById('summary-pizza-prezzo').innerText = `€ ${pizza.prezzoVendita.toFixed(2)}`;
            document.getElementById('summary-pizza-costo').innerText = `€ ${costoTotale.toFixed(2)}`;
            document.getElementById('summary-pizza-margine').innerText = `€ ${margine.toFixed(2)}`;
            document.getElementById('summary-pizza-fc').innerText = `${fcPerc.toFixed(1)}%`;

            pizza.ingredienti.forEach((ing, idx) => {
                const mp = materiePrime.find(m => m.id === ing.mpId);
                if (!mp) return;

                const costoSingolo = getCostoIngredienteInPizza(ing.mpId, ing.qta);
                const tr = document.createElement('tr');
                tr.innerHTML = `
                    <td class="p-3 text-white font-medium">${mp.nome}</td>
                    <td class="p-3">${ing.qta} g/ml</td>
                    <td class="p-3">€ ${getCostoNettoAlKgL(mp).toFixed(2)} / ${mp.Unità}</td>
                    <td class="p-3 font-bold text-orange-400">€ ${costoSingolo.toFixed(2)}</td>
                    <td class="p-3 text-center">
                        <button onclick="rimuoviIngredienteDaPizza(${idx})" class="text-red-400 hover:text-red-300"><i class="fa-solid fa-minus-circle"></i></button>
                    </td>
                `;
                tbody.appendChild(tr);
            });
        }

        function aggiungiIngredienteAPizza() {
            if (!selectedPizzaId) return;
            const pizza = pizze.find(p => p.id === selectedPizzaId);
            const mpId = parseInt(document.getElementById('add-ingrediente-select').value);
            const qta = parseFloat(document.getElementById('add-ingrediente-qta').value);

            if (!qta || qta <= 0) {
                Swal.fire('Attenzione', 'Inserisci una quantità valida in grammi o ml.', 'warning');
                return;
            }

            pizza.ingredienti.push({ mpId, qta });
            document.getElementById('add-ingrediente-qta').value = '';
            saveData();
        }

        function rimuoviIngredienteDaPizza(index) {
            const pizza = pizze.find(p => p.id === selectedPizzaId);
            if (pizza) {
                pizza.ingredienti.splice(index, 1);
                saveData();
            }
        }

        // 3. MENU ENGINEERING MATRIX LOGIC
        function getMatrixCategory(pizza, margineMedio, venditeMedie) {
            const costo = getCostoTotalePizza(pizza);
            const margine = pizza.prezzoVendita - costo;
            const popolarita = pizza.venditeMensili;

            if (margine >= margineMedio && popolarita >= venditeMedie) return { cat: "STAR", color: "bg-emerald-500/20 text-emerald-400 border-emerald-500", cons: "Mantieni la ricetta alta qualità, promuovi al massimo." };
            if (margine < margineMedio && popolarita >= venditeMedie) return { cat: "WORKHORSE", color: "bg-blue-500/20 text-blue-400 border-blue-500", cons: "Aumenta leggermente il prezzo o riduci i grammi degli ingredienti costosi." };
            if (margine >= margineMedio && popolarita < venditeMedie) return { cat: "PUZZLE", color: "bg-amber-500/20 text-amber-400 border-amber-500", cons: "Migliora la visibilità nel menù o suggeriscila tramite i camerieri." };
            return { cat: "DOG", color: "bg-red-500/20 text-red-400 border-red-500", cons: "Valuta di rimuoverla o riprogettarla completamente." };
        }

        function renderMenuEngineeringTable() {
            const tbody = document.getElementById('tbl-engineering-body');
            tbody.innerHTML = '';

            if (pizze.length === 0) return;

            const margineTotale = pizze.reduce((acc, p) => acc + (p.prezzoVendita - getCostoTotalePizza(p)), 0);
            const margineMedio = margineTotale / pizze.length;

            const venditeTotali = pizze.reduce((acc, p) => acc + p.venditeMensili, 0);
            const venditeMedie = venditeTotali / pizze.length;

            pizze.forEach(p => {
                const costo = getCostoTotalePizza(p);
                const margine = p.prezzoVendita - costo;
                const fcPerc = p.prezzoVendita > 0 ? (costo / p.prezzoVendita) * 100 : 0;
                const matrix = getMatrixCategory(p, margineMedio, venditeMedie);

                const tr = document.createElement('tr');
                tr.className = "hover:bg-slate-800/50";
                tr.innerHTML = `
                    <td class="p-3 font-semibold text-white">${p.nome}</td>
                    <td class="p-3">€ ${p.prezzoVendita.toFixed(2)}</td>
                    <td class="p-3 text-rose-400">€ ${costo.toFixed(2)}</td>
                    <td class="p-3 text-emerald-400 font-bold">€ ${margine.toFixed(2)}</td>
                    <td class="p-3">${fcPerc.toFixed(1)}%</td>
                    <td class="p-3"><span class="px-2.5 py-1 rounded-full border text-xs font-bold ${matrix.color}">${matrix.cat}</span></td>
                    <td class="p-3 text-xs text-slate-300">${matrix.cons}</td>
                `;
                tbody.appendChild(tr);
            });
        }

        // 4. DASHBOARD & GRAFICI
        function renderDashboard() {
            document.getElementById('dash-tot-pizze').innerText = pizze.length;
            
            if (pizze.length === 0) return;

            let totFC = 0;
            let totMargine = 0;
            let starCount = 0;

            const margineTotale = pizze.reduce((acc, p) => acc + (p.prezzoVendita - getCostoTotalePizza(p)), 0);
            const margineMedio = margineTotale / pizze.length;
            const venditeMedie = pizze.reduce((acc, p) => acc + p.venditeMensili, 0) / pizze.length;

            pizze.forEach(p => {
                const costo = getCostoTotalePizza(p);
                const margine = p.prezzoVendita - costo;
                totFC += (p.prezzoVendita > 0 ? (costo / p.prezzoVendita) * 100 : 0);
                totMargine += margine;

                const cat = getMatrixCategory(p, margineMedio, venditeMedie).cat;
                if(cat === 'STAR') starCount++;
            });

            document.getElementById('dash-fc-medio').innerText = `${(totFC / pizze.length).toFixed(1)}%`;
            document.getElementById('dash-margine-medio').innerText = `€ ${(totMargine / pizze.length).toFixed(2)}`;
            document.getElementById('dash-star-count').innerText = starCount;

            renderCharts(margineMedio, venditeMedie);
        }

        function renderCharts(margineMedio, venditeMedie) {
            // Chart 1: Matrix
            const ctx1 = document.getElementById('chartMatrix').getContext('2d');
            if (chartMatrixInstance) chartMatrixInstance.destroy();

            const categoriesCount = { STAR: 0, WORKHORSE: 0, PUZZLE: 0, DOG: 0 };
            pizze.forEach(p => {
                const cat = getMatrixCategory(p, margineMedio, venditeMedie).cat;
                categoriesCount[cat]++;
            });

            chartMatrixInstance = new Chart(ctx1, {
                type: 'bar',
                data: {
                    labels: ['STAR', 'WORKHORSE', 'PUZZLE', 'DOG'],
                    datasets: [{
                        label: 'Numero Pizze',
                        data: [categoriesCount.STAR, categoriesCount.WORKHORSE, categoriesCount.PUZZLE, categoriesCount.DOG],
                        backgroundColor: ['#10b981', '#3b82f6', '#f59e0b', '#ef4444']
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: { legend: { display: false } },
                    scales: { y: { ticks: { stepSize: 1, color: '#94a3b8' } }, x: { ticks: { color: '#94a3b8' } } }
                }
            });

            // Chart 2: Top Food Cost
            const ctx2 = document.getElementById('chartFoodCost').getContext('2d');
            if (chartFCInstance) chartFCInstance.destroy();

            const sortedPizze = [...pizze].sort((a,b) => getCostoTotalePizza(b) - getCostoTotalePizza(a)).slice(0, 5);

            chartFCInstance = new Chart(ctx2, {
                type: 'doughnut',
                data: {
                    labels: sortedPizze.map(p => p.nome),
                    datasets: [{
                        data: sortedPizze.map(p => getCostoTotalePizza(p).toFixed(2)),
                        backgroundColor: ['#f97316', '#06b6d4', '#8b5cf6', '#ec4899', '#64748b']
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: { legend: { position: 'bottom', labels: { color: '#e2e8f0' } } }
                }
            });
        }
